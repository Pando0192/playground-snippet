// T54 - DATA PROPERTY DUMP (read-only: reads and prints the VALUES of
// the non-function properties on window.oai; calls NO functions).
(function () {
  var d = document;
  var box = d.createElement("div");
  box.id = "t54-box";
  box.style.cssText = "position:fixed;left:12px;bottom:12px;z-index:2147483647;font:12px Menlo,monospace;color:#ff5252;background:rgba(0,0,0,.85);border:1px solid #ff5252;border-radius:6px;padding:8px 10px;max-width:80vw;max-height:50vh;overflow:auto;white-space:pre-wrap;word-break:break-all";
  d.body.appendChild(box);

  function show(v) {
    try {
      if (v === null) return "null";
      var s = typeof v === "object" ? JSON.stringify(v) : String(v);
      if (s === undefined) s = String(v);
      if (s.length > 400) s = s.slice(0, 400) + "...<truncated " + s.length + " chars>";
      return "(" + typeof v + ") " + s;
    } catch (e) {
      return "(unreadable: " + e + ")";
    }
  }

  var api = window.oai || window.openai || window.webplus;
  var lines = [];
  if (!api) {
    lines.push("no API object found");
  } else {
    var keys = Object.keys(api);
    for (var k = 0; k < keys.length; k++) {
      var key = keys[k];
      try {
        var v = api[key];
        if (typeof v === "function") continue; // functions not called
        lines.push(key + " = " + show(v));
      } catch (e) {
        lines.push(key + " = (threw: " + e + ")");
      }
    }
  }
  var out = "T54 DATA DUMP (no calls made):\n" + lines.join("\n");
  box.textContent = out;
  try { console.log(out); } catch (e) {}
})();
