/* T61 FETCH CONTROL - Pluto Security research, inert canary.
   Question: can the VIEWER-session widget fetch a PINNED jsdelivr URL?
   (The playground auto-load says "Failed to fetch cdn.jsdelivr.net...",
   but the captured widget CSP allowlists jsdelivr for connect-src.)
   A = pinned commit 6f89b79 (immutable). B = @main (mutable ref).
   This is INBOUND only - no data leaves the victim. */
(function () {
  function log(m) { try { console.log("[T61] " + m); } catch (e) {} }
  log("START fetch control (inert, inbound-only) " + Date.now());

  var BASE = "https://cdn.jsdelivr.net/gh/Pando0192/playground-snippet";
  var A = BASE + "@6f89b79/snippet.js"; /* pinned */
  var B = BASE + "@main/snippet.js";    /* mutable */

  function tryFetch(tag, url) {
    return fetch(url, { cache: "no-store" })
      .then(function (r) {
        return r.text().then(function (t) {
          log(tag + " FETCH-OK status=" + r.status + " len=" + t.length +
              " head=" + JSON.stringify(t.slice(0, 40)));
        });
      })
      .catch(function (e) {
        log(tag + " FETCH-FAIL name=" + (e && e.name) +
            " msg=" + (e && e.message));
      });
  }

  tryFetch("A-pinned", A)
    .then(function () { return tryFetch("B-main", B); })
    .then(function () { log("DONE A and B attempted"); })
    .catch(function (e) { log("UNEXPECTED " + e); });
})();
