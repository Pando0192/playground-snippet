// T49e - allowlist mapping: is google.com gate-safe? (Pluto Security research canary)
// Only example.com is known gate-safe (T33, re-confirmed T49c). This probe
// answers whether the server-side allowlist includes top sites at all.
// If google.com is silent -> the list includes top sites -> user-content
// pages on trusted domains (blogspot etc.) become the next candidates.
// If dialog -> the list is a tiny curated set -> honest ceiling stands.
// Inert: opens the google.com homepage, nothing else.
(function () {
  try {
    Object.defineProperty(navigator, "userActivation", {
      get: function () { return { isActive: true, hasBeenActive: true }; },
      configurable: true
    });
  } catch (e) {}

  var URL_T49E = "https://www.google.com/";

  var d = document;
  var box = d.createElement("div");
  box.id = "t49e-box";
  box.style.cssText = "position:fixed;left:12px;bottom:12px;z-index:2147483647;font:12px Menlo,monospace;color:#ff5252;background:rgba(0,0,0,.85);border:1px solid #ff5252;border-radius:6px;padding:8px 10px;max-width:80vw;white-space:pre-wrap";
  box.textContent = "T49e (inert) - firing google.com in 1s...";
  d.body.appendChild(box);

  setTimeout(function () {
    var api = window.oai || window.openai || window.webplus;
    if (!api || !api.openExternal) {
      box.textContent = "T49e: no oai.openExternal API found";
      return;
    }
    try {
      api.openExternal({ href: URL_T49E });
      box.textContent =
        "T49e FIRED: https://www.google.com/\n\n" +
        "silent tab = top sites are on the allowlist\n" +
        "dialog = allowlist is a tiny curated set";
    } catch (e) {
      box.textContent = "T49e threw: " + e;
    }
  }, 1000);
})();
