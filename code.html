// T53 - LIVE API SURFACE ENUMERATION (pure read-only: lists the
// API object's own keys and their types; calls NOTHING).
(function () {
  var d = document;
  var box = d.createElement("div");
  box.id = "t53-box";
  box.style.cssText = "position:fixed;left:12px;bottom:12px;z-index:2147483647;font:12px Menlo,monospace;color:#ff5252;background:rgba(0,0,0,.85);border:1px solid #ff5252;border-radius:6px;padding:8px 10px;max-width:80vw;max-height:40vh;overflow:auto;white-space:pre-wrap;word-break:break-all";
  d.body.appendChild(box);

  var names = ["oai", "openai", "webplus"];
  var lines = [];
  var api = null;
  for (var i = 0; i < names.length; i++) {
    try {
      var cand = window[names[i]];
      if (cand && typeof cand === "object") {
        api = cand;
        lines.push("window." + names[i] + " EXISTS");
        break;
      }
    } catch (e) {}
  }
  if (!api) {
    lines.push("no API object on window (oai/openai/webplus)");
  } else {
    var keys = Object.keys(api);
    lines.push("API keys (" + keys.length + "):");
    for (var k = 0; k < keys.length; k++) {
      var t;
      try { t = typeof api[keys[k]]; } catch (e) { t = "unreadable"; }
      lines.push("  " + keys[k] + " : " + t);
    }
    // also any nested sub-objects (one level)
    for (var k2 = 0; k2 < keys.length; k2++) {
      try {
        var v = api[keys[k2]];
        if (v && typeof v === "object") {
          var sub = Object.keys(v);
          if (sub.length) lines.push("  ." + keys[k2] + " sub-keys: " + sub.join(","));
        }
      } catch (e) {}
    }
  }
  var out = "T53 API SURFACE (no calls made):\n" + lines.join("\n");
  box.textContent = out;
  try { console.log(out); } catch (e) {}
})();
