// T49f - allowlist mapping: is blogspot.com gate-safe? (Pluto Security research canary)
// google.com is silent (T49e) - the allowlist includes top sites. blogspot.com
// (Blogger) is the standout top-tier domain where USERS control JavaScript
// (free custom themes). This probes the gate with a famous blogspot URL
// (Google's official blog) BEFORE we invest in creating a researcher-owned
// blog for the full chain test. Inert: opens Google's blog homepage.
(function () {
  try {
    Object.defineProperty(navigator, "userActivation", {
      get: function () { return { isActive: true, hasBeenActive: true }; },
      configurable: true
    });
  } catch (e) {}

  var URL_T49F = "https://googleblog.blogspot.com/";

  var d = document;
  var box = d.createElement("div");
  box.id = "t49f-box";
  box.style.cssText = "position:fixed;left:12px;bottom:12px;z-index:2147483647;font:12px Menlo,monospace;color:#ff5252;background:rgba(0,0,0,.85);border:1px solid #ff5252;border-radius:6px;padding:8px 10px;max-width:80vw;white-space:pre-wrap";
  box.textContent = "T49f (inert) - firing googleblog.blogspot.com in 1s...";
  d.body.appendChild(box);

  setTimeout(function () {
    var api = window.oai || window.openai || window.webplus;
    if (!api || !api.openExternal) {
      box.textContent = "T49f: no oai.openExternal API found";
      return;
    }
    try {
      api.openExternal({ href: URL_T49F });
      box.textContent =
        "T49f FIRED: https://googleblog.blogspot.com/\n\n" +
        "silent tab = blogspot.com is gate-safe -> full chain test next\n" +
        "dialog = blogspot.com is off the allowlist";
    } catch (e) {
      box.textContent = "T49f threw: " + e;
    }
  }, 1000);
})();
