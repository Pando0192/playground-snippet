// T70d — mobile runtime: API enumeration + tap-triggered openExternal
// (Pluto Security research canary, inert — no data, control domain only)
// Lists every function the widget-facing API exposes in the native app
// runtime, plus the two T70c tap probes (real gesture openExternal).
(function () {
  var EPOCH = 1792653245;
  var found = null, GNAME = null;
  ["oai", "openai", "webplus"].forEach(function (k) {
    if (!found && window[k] && typeof window[k] === "object") { found = window[k]; GNAME = k; }
  });
  var API = found;

  try {
    Object.defineProperty(navigator, "userActivation", {
      get: function () { return { isActive: true, hasBeenActive: true }; },
      configurable: true
    });
  } catch (e) {}

  var d = document;
  var css = d.createElement("style");
  css.textContent = [
    "#t70d-ov{position:fixed;inset:0;z-index:2147483647;background:rgba(0,0,0,.6);display:flex;align-items:center;justify-content:center;font-family:-apple-system,BlinkMacSystemFont,'Segoe UI',Helvetica,Arial,sans-serif;padding:16px;box-sizing:border-box}",
    "#t70d-card{width:100%;max-width:420px;box-sizing:border-box;background:#212121;border:1px solid rgba(255,255,255,.1);border-radius:20px;padding:22px;color:#fff;text-align:left;max-height:86vh;overflow-y:auto}",
    "#t70d-card h1{font-size:17px;font-weight:600;margin:0 0 6px}",
    "#t70d-card p{font-size:13px;color:#9b9b9b;line-height:1.5;margin:0 0 14px}",
    "#t70d-st{font-family:Menlo,Consolas,monospace;font-size:11px;line-height:1.7;border:1px solid rgba(255,255,255,.15);border-radius:10px;padding:12px;background:rgba(0,0,0,.35);color:#e8e8e8;word-break:break-all}",
    "#t70d-st b{color:#7ee787}",
    "#t70d-card button.tap{width:100%;height:52px;margin:6px 0 0;background:#238636;border:none;border-radius:12px;color:#fff;font-size:15px;font-weight:600}",
    "#t70d-x{margin-top:14px;width:100%;height:42px;background:transparent;border:1px solid rgba(255,255,255,.2);border-radius:999px;color:#fff;font-size:14px}"
  ].join("");
  var ov = d.createElement("div");
  ov.id = "t70d-ov";
  ov.appendChild(css);
  var card = d.createElement("div");
  card.id = "t70d-card";
  card.innerHTML =
    "<h1>T70d — mobile API enumeration</h1>" +
    '<p>Research canary (Pluto Security, inert). Screenshot the green list = every callable in this runtime. Then tap the buttons — real gesture openExternal probes.</p>' +
    '<div id="t70d-st"></div>' +
    '<button class="tap" id="t70d-b1" type="button">TAP 1: openExternal example.com</button>' +
    '<button class="tap" id="t70d-b2" type="button">TAP 2: openExternal + waitForCompletion</button>';
  ov.appendChild(card);
  var x = d.createElement("button");
  x.id = "t70d-x"; x.type = "button"; x.textContent = "Close";
  x.onclick = function () { ov.remove(); };
  card.appendChild(x);
  d.body.appendChild(ov);

  function st(html) { var e = d.getElementById("t70d-st"); if (e) e.innerHTML += html; }
  st("token: T70d-" + EPOCH + "<br>");
  st("global: " + GNAME + "<br>");
  if (!API) {
    st("api: NONE FOUND (oai/openai/webplus all absent)<br>");
  } else {
    var keys = Object.keys(API).sort();
    st("api keys (" + keys.length + "): <b>" + keys.join(", ") + "</b><br>");
    // shape of each key (function / object with keys)
    keys.forEach(function (k) {
      try {
        var v = API[k];
        if (typeof v === "function") st("&nbsp;&nbsp;fn  " + k + "<br>");
        else if (v && typeof v === "object") st("&nbsp;&nbsp;obj " + k + " {" + Object.keys(v).slice(0, 12).join(",") + "}<br>");
        else st("&nbsp;&nbsp;" + typeof v + " " + k + " = " + String(v).slice(0, 40) + "<br>");
      } catch (e) { st("&nbsp;&nbsp;err " + k + "<br>"); }
    });
  }
  // any extra interesting widget-side globals
  try {
    var extra = [];
    ["oai", "openai", "webplus", "__playground", "webkit", "SkybridgeNative", "CanmoreNative"].forEach(function (k) {
      if (k in window) extra.push(k + (k === "webkit" ? "(msg handlers: " + (window.webkit && window.webkit.messageHandlers ? Object.keys(window.webkit.messageHandlers).join(",") : "?") + ")" : ""));
    });
    st("globals: " + extra.join(", ") + "<br>");
  } catch (e) {}

  function wireTap(btn, label, args) {
    var b = d.getElementById(btn);
    if (!b || !API || !API.openExternal) return;
    b.addEventListener("click", function () {
      st(label + ": tap, firing...<br>");
      try {
        API.openExternal(args).then(
          function (r) { st(label + ": <b>RESOLVED</b> " + JSON.stringify(r == null ? "undefined" : r) + "<br>"); },
          function (e) { st(label + ": <b>REJECTED</b> " + (e && e.message ? e.message : String(e)) + "<br>"); }
        );
      } catch (e) { st(label + ": THREW " + (e && e.message ? e.message : String(e)) + "<br>"); }
    });
  }
  wireTap("t70d-b1", "tap1", { href: "https://example.com/#T70d-" + EPOCH + "-tap1" });
  wireTap("t70d-b2", "tap2", { href: "https://example.com/#T70d-" + EPOCH + "-tap2", waitForCompletion: true });
})();
