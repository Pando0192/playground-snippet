// T49d - does the server-side gate resolve redirects? (Pluto Security research canary)
// tinyurl.com/yqp7ct redirects to https://example.com/ (the known server-SAFE
// destination). If the gate checks only the PRESENTED host (tinyurl.com), this
// dialogs (shortener not trusted). If the gate FOLLOWS the redirect and judges
// the FINAL destination (example.com), this opens silently - proving the
// checker resolves URLs server-side, which opens a new attack surface
// (checker-fetch exfiltration / UA-differential redirects). Inert: opens
// example.com at most.
(function () {
  try {
    Object.defineProperty(navigator, "userActivation", {
      get: function () { return { isActive: true, hasBeenActive: true }; },
      configurable: true
    });
  } catch (e) {}

  var URL_T49D = "https://tinyurl.com/yqp7ct";

  var d = document;
  var box = d.createElement("div");
  box.id = "t49d-box";
  box.style.cssText = "position:fixed;left:12px;bottom:12px;z-index:2147483647;font:12px Menlo,monospace;color:#ff5252;background:rgba(0,0,0,.85);border:1px solid #ff5252;border-radius:6px;padding:8px 10px;max-width:80vw;white-space:pre-wrap";
  box.textContent = "T49d (inert) - firing tinyurl->example.com in 1s...";
  d.body.appendChild(box);

  setTimeout(function () {
    var api = window.oai || window.openai || window.webplus;
    if (!api || !api.openExternal) {
      box.textContent = "T49d: no oai.openExternal API found";
      return;
    }
    try {
      api.openExternal({ href: URL_T49D });
      box.textContent =
        "T49d FIRED: tinyurl.com/yqp7ct (-> example.com)\n\n" +
        "silent tab = gate FOLLOWS redirects (judges final destination)\n" +
        "dialog = gate judges the presented host only";
    } catch (e) {
      box.textContent = "T49d threw: " + e;
    }
  }, 1000);
})();
